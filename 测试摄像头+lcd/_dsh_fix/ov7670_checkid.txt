

/*===================读取器件ID，用来验证SCCB是否真的通了===================
	OV7670 的 PID(0x0A) 读出来固定是 0x76，VER(0x0B) 读出来固定是 0x73。
	如果这两个值读不对，说明 SCCB 根本没写进摄像头，后面写的整张寄存器表
	等于全部作废，摄像头还停在上电默认的 YUV 输出，画面必然是乱的。
	返回值：0 = SCCB正常；1 = SCCB不通
========================================================================*/
uint8_t OV7670_CheckID(void)
{
	uint8_t pid;
	uint8_t ver;

	pid = SCCB_ReadReg(0x0A);
	ver = SCCB_ReadReg(0x0B);

	if ((pid == 0x76) && (ver == 0x73))
	{
		return 0;
	}
	return 1;
}
