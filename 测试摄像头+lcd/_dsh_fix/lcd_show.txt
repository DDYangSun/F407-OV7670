/*==========================================================================
	函数说明：把 OV7670 的一帧 QQVGA(160x120) RGB565 数据显示到 ST7735 上
	说明：
	  1) 必须先调用 LCD_Address_Set() 设定窗口！
	     原来的写法是直接发 0x2C，地址指针还停在上一次(比如 LCD_Fill)写完的
	     位置，数据会从错误的行列开始填，画面必然错位/花屏。
	  2) 屏幕是 128x160，摄像头一行有 160 个点，一行写不下 160 个点，
	     所以每行只取中间的 128 个点(左右各裁掉 16 个)，画面才不会被拉斜。
==========================================================================*/
void LCD_ShowOV7670PIC(uint16_t *Image)
{
	uint16_t x, y;
	uint16_t camw = 160;						//	摄像头输出宽度 QQVGA=160
	uint16_t startx = (camw - LCD_W) / 2;		//	(160-128)/2 = 16

	LCD_Address_Set(0, 0, LCD_W - 1, 119);		//	开一个 128 x 120 的窗口

	for (y = 0; y < 120; y++)
	{
		for (x = 0; x < LCD_W; x++)
		{
			LCD_WR_DATA(Image[y * camw + startx + x]);
		}
	}
}
