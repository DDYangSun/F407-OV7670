#include "stm32f4xx.h"
#include "Delay.h"
#include "lcd.h"
#include "pic.h"
#include "ov7670.h"
#include "SCCB.h"
#include "stm32f4xx_conf.h"

extern uint16_t QQVGA_Frame[];		//储存原始画面0
extern uint16_t QQVGA_Frame1[];		//原始画面的位置1

uint16_t *Read_Frame = QQVGA_Frame;
volatile uint8_t LCD_ShowReady = 0;	//	等第一帧到了再显示

volatile uint32_t frame_cnt = 0;	//	DMA 完成一帧的次数
volatile uint32_t ovr_cnt   = 0;	//	DCMI 上溢(OVR)次数
volatile uint32_t rearm_cnt = 0;	//	重新武装采集的次数
uint32_t draw_cnt = 0;				//	实际刷新到屏上的帧数

/*	把一字节按两位十六进制显示出来，用于把读到的 ID 直接打到屏上	*/
static void LCD_ShowHex2(u16 x, u16 y, uint8_t v, u16 fc, u16 bc)
{
	char s[3];
	const char *hex = "0123456789ABCDEF";

	s[0] = hex[(v >> 4) & 0x0F];
	s[1] = hex[v & 0x0F];
	s[2] = '\0';
	LCD_ShowString(x, y, (const u8 *)s, fc, bc, 16, 0);
}

/*	把十进制数按固定宽度（不足补 0）写进字符串，返回新的写入位置	*/
static u8 *PutU32(u8 *p, uint32_t v, u8 width)
{
	u8 buf[10];
	u8 n = 0;

	if (v == 0)
	{
		buf[n++] = '0';
	}
	while ((v != 0) && (n < sizeof(buf)))
	{
		buf[n++] = (u8)('0' + (v % 10));
		v /= 10;
	}
	while (n < width)
	{
		buf[n++] = '0';
	}
	while (n > 0)
	{
		*p++ = buf[--n];
	}
	return p;
}

/*	在画面下方空白处显示统计：
		F = DMA 收到新帧的次数（画面一直在动，这个数就会一直涨）
		D = 真正刷到屏上的帧数
		O = DCMI 上溢(OVR)次数
		R = 重新武装采集的次数
	文字还在变、只有画面定格  -> 摄像头/DMA 这一层停了
	文字和画面一起定格        -> 程序本身卡死（死机/HardFault）	*/
static void LCD_ShowStat(void)
{
	char s[24];
	u8 *p;

	p = s;
	*p++ = 'F'; p = PutU32(p, frame_cnt, 5);
	*p++ = 'D'; p = PutU32(p, draw_cnt, 5);
	*p = '\0';
	LCD_ShowString(0, 124, (const u8 *)s, BLACK, WHITE, 16, 1);

	p = s;
	*p++ = 'O'; p = PutU32(p, ovr_cnt, 3);
	*p++ = 'R'; p = PutU32(p, rearm_cnt, 2);
	*p = '\0';
	LCD_ShowString(0, 140, (const u8 *)s, BLACK, WHITE, 16, 1);
}

int main(void)
{
	uint32_t last_frame = 0;
	uint32_t idle = 0;
	uint32_t tick = 0;

	/*--------------------------- 第0步：延时函数初始化 --------------------------
		Delay_Init 的参数是"系统时钟频率，单位 MHz"。
		Delay.c 里把 SysTick 的时钟设成 HCLK/8，再算 fac_us = SYSCLK/8，
		即"1 微秒需要多少个 SysTick 计数"。
		必须和真实主频一致，否则延时整体偏大或偏小；
		填 0 会让延时全部变成空操作（之前 SCCB 读不到 ID 就是这个原因）。
		这里直接用 SystemCoreClock（system_stm32f4xx.c 里算好的实际频率）换算，
		比硬写 168 更保险。														*/
	Delay_Init((uint8_t)(SystemCoreClock / 1000000));

	/*--------------------------- 第1步：先把 LCD 点亮 ---------------------------*/
	LCD_Init();
	LCD_Fill(0, 0, LCD_W, LCD_H, WHITE);

	/*--------------------------- 第2步：配置 DCMI 用到的 GPIO ------------------*/
	OV7670_DCMI_Init();

	/*--------------------------- 第3步：给时钟 + 硬件复位 ----------------------*/
	OV7670_MCLK_start();		//	PA8 输出 8MHz XCLK
	Delay_ms(10);
	OV7670_Rest();				//	PA1=PWDN(低=工作)，PA3=RST
	Delay_ms(20);

	/*--------------------------- 第4步：SCCB 自检 ------------------------------*/
	SCCB_Init();				//	SCL=PB10，SDA=PB11
	Delay_ms(10);

	{
		uint8_t pid, ver, line;

		line = SCCB_TestLine();		//	0=正常 1=SDA一直低 2=SDA一直高
		pid  = SCCB_ReadReg(0x0A);	//	OV7670 的 PID 固定 0x76
		ver  = SCCB_ReadReg(0x0B);	//	OV7670 的 VER 固定 0x73

		if ((line != 0) || (pid != 0x76) || (ver != 0x73))
		{
			LCD_Fill(0, 0, LCD_W, LCD_H, WHITE);
			LCD_ShowString(0,  0, (const u8 *)"SCCB ERROR", RED, WHITE, 16, 0);
			LCD_ShowString(0, 20, (const u8 *)"PID=", BLACK, WHITE, 16, 0);
			LCD_ShowHex2(32, 20, pid, BLACK, WHITE);
			LCD_ShowString(0, 40, (const u8 *)"VER=", BLACK, WHITE, 16, 0);
			LCD_ShowHex2(32, 40, ver, BLACK, WHITE);
			LCD_ShowString(0, 60, (const u8 *)"LINE=", BLACK, WHITE, 16, 0);
			LCD_ShowHex2(40, 60, line, BLACK, WHITE);
			LCD_ShowString(0, 90, (const u8 *)"SCL=PB10", BLACK, WHITE, 16, 0);
			LCD_ShowString(0, 110, (const u8 *)"SDA=PB11", BLACK, WHITE, 16, 0);
			while (1);
		}
	}

	/*--------------------------- 第5步：写寄存器表（RGB565 + QVGA）-------------*/
	OV7670_RegConfig();

	/*--------------------------- 第6步：设置采集窗口 ---------------------------*/
	OV7670_Window_Set(160, 120, 160, 120);

	/*--------------------------- 第7步：DMA + DCMI 开始采集 -------------------*/
	OV7670DMA_DCMItoSRAM();
	OV7670_GetPicture();

	LCD_Fill(0, 0, LCD_W, LCD_H, BLACK);

	while(1)
	{
		/*	1. 有新帧就刷屏	*/
		if (LCD_ShowReady)
		{
			LCD_ShowReady = 0;
			LCD_ShowOV7670PIC(Read_Frame);
			draw_cnt++;
		}

		/*	2. 统计 DCMI 上溢(OVR)：一旦上溢，DCMI 硬件会自动停掉 CAPTURE，
			   表现就是"画面定格、再也不刷新"。这里把标志清掉并计数	*/
		if (DCMI_GetFlagStatus(DCMI_FLAG_OVFRI) != RESET)
		{
			DCMI_ClearFlag(DCMI_FLAG_OVFRI);
			ovr_cnt++;
		}

		/*	3. 兜底：连续约 500ms 没有新帧，就重新武装 DCMI + DMA，
			   不管是上溢、DMA 传输错误还是别的原因，都能自己爬起来	*/
		if (frame_cnt != last_frame)
		{
			last_frame = frame_cnt;
			idle = 0;
		}
		else
		{
			Delay_ms(10);
			if (++idle >= 50)
			{
				idle = 0;
				rearm_cnt++;

				OV7670_StopGetPicture();
				OV7670DMA_DCMItoSRAM();
				Read_Frame = QQVGA_Frame;
				LCD_ShowReady = 0;
				OV7670_GetPicture();
			}
		}

		/*	4. 每 100 次循环刷新一次统计信息	*/
		if (++tick >= 100)
		{
			tick = 0;
			LCD_ShowStat();
		}
	}
}


void DMA2_Stream1_IRQHandler(void)
{
	if (DMA_GetITStatus(DMA2_Stream1, DMA_IT_TCIF1) == SET)	//	DMA 传输完成
	{
		if (DMA_GetCurrentMemoryTarget(DMA2_Stream1) == SET)	//	CT=1：接下来写缓冲区1，说明缓冲区0已满
		{
			Read_Frame = QQVGA_Frame;		//	缓冲区0 可以读了
		}
		else
		{
			Read_Frame = QQVGA_Frame1;		//	缓冲区1 可以读了
		}
		frame_cnt++;
		LCD_ShowReady = 1;

		DMA_ClearITPendingBit(DMA2_Stream1, DMA_IT_TCIF1);	//	清中断标志
	}
}
